const SHEET_NAME = 'Income';
const HEADERS = ['ID','Date','Month','Category','Source','Reference','Gross','Charges','Net','Note'];

function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) { sh = ss.insertSheet(SHEET_NAME); }
  if (sh.getLastRow() === 0) {
    sh.appendRow(HEADERS);
    sh.setFrozenRows(1);
    sh.getRange('G:I').setNumberFormat('#,##0.00');
  }
  return sh;
}

function out_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  return handle_(e.parameter.action || 'list', e.parameter.payload || '');
}

function doPost(e) {
  let body = {};
  try { body = JSON.parse(e.postData.contents); } catch (err) { body = {}; }
  const payload = body.payload ? JSON.stringify(body.payload) : '';
  return handle_(body.action || 'list', payload);
}

function handle_(action, payloadStr) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = getSheet_();
    const data = payloadStr ? JSON.parse(payloadStr) : null;
    if (action === 'add') { writeRows_(sh, [data]); }
    if (action === 'bulk') { writeRows_(sh, data); }
    if (action === 'update') { deleteRow_(sh, data.id); writeRows_(sh, [data]); }
    if (action === 'delete') { deleteRow_(sh, data.id); }
    return out_({ ok: true, rows: readRows_(sh) });
  } catch (err) {
    return out_({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function readRows_(sh) {
  const last = sh.getLastRow();
  if (last < 2) { return []; }
  const vals = sh.getRange(2, 1, last - 1, HEADERS.length).getValues();
  return vals.filter(function (r) { return r[0] !== ''; }).map(function (r) {
    return {
      id: String(r[0]), date: fmtDate_(r[1]), category: String(r[3]),
      source: String(r[4]), ref: String(r[5]),
      gross: Number(r[6]) || 0, charges: Number(r[7]) || 0,
      net: Number(r[8]) || 0, note: String(r[9])
    };
  });
}

function fmtDate_(v) {
  if (v instanceof Date) { return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); }
  return String(v).slice(0, 10);
}

function writeRows_(sh, list) {
  const out = list.map(function (d) {
    return [d.id, d.date, String(d.date).slice(0, 7), d.category, d.source,
            d.ref, Number(d.gross) || 0, Number(d.charges) || 0,
            Number(d.net) || 0, d.note];
  });
  if (out.length) { sh.getRange(sh.getLastRow() + 1, 1, out.length, HEADERS.length).setValues(out); }
}

function deleteRow_(sh, id) {
  const last = sh.getLastRow();
  if (last < 2) { return; }
  const ids = sh.getRange(2, 1, last - 1, 1).getValues();
  for (let i = ids.length - 1; i >= 0; i--) {
    if (String(ids[i][0]) === String(id)) { sh.deleteRow(i + 2); }
  }
}
